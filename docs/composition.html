tidyall <- function(m0) {
  r <- bind_rows(
    m0 %>% anova() %>% tidy,
    m0 %>% ranova() %>% tidy,
    m0 %>% tidy)
  return(r)
}

pc <- 1e-5
tb16s <- phyX %>% 
  subset_samples(!str_detect(Sample.name,'neg')) %>% 
  subset_samples(!str_detect(Sample.name,'poz')) %>% 
  filter_taxa(function(x) sum(x>0)>4, TRUE) %>% 
  transform_sample_counts(function(x) x / sum(x)) %>% 
  psmelt() %>% data.frame() %>% 
  group_by(ta3,ta4,ta5,ta6,ta7,OTU) %>% 
  do(lmer(data = ., log(Abundance + pc) ~ factor(day) + (1|Participant) + (1|cycle_part))  %>% tidyall())

fmrt <- list(
  # estimate = colDef(format = colFormat(scientific = TRUE, digits = 2)),
  estimate = colDef(
    cell = function(value) {
      if (is.na(value)) return("")
      formatC(value, format = "e", digits = 2)
    }
  ),
  
  std.error = colDef(
    cell = function(value) {
      if (is.na(value)) return("")
      formatC(value, format = "e", digits = 2)
    }
  ),
  
  p.value = colDef(
    cell = function(value) {
      if (is.na(value)) return("")
      formatC(value, format = "e", digits = 2)
    }
  )
)
